% parameters (x,y) center coords, theta vertex angle, rho radius in cm.

\draw[red] (#1,#2) + (#3:#4cm) -- + (#3+72:#4cm) -- 
+(#3+144:#4cm) -- +(#3+216:#4cm) -- + (#3+288:#4cm) -- cycle
}
